# ============================================================
#  PIP PACKAGES
# ============================================================

openai-whisper
# Speech-to-text conversion. Converts patient recordings to text strings. 
# Processes on the Mac's M2 GPU using the MPS backend (fp16=False).

flask
# Web API framework hosting /ping, /transcribe, /next_question, and /flag_urgent.

requests
# Handles local network API requests directed to Ollama.

torch
# Deep learning wrapper providing backend orchestration for Whisper.

torchvision
# Required alongside torch for framework environment stability.

torchaudio
# Critical for handling tensor transforms on Whisper's incoming wave streams.

numpy
# Core mathematical array library used extensively across Whisper data frames.

# ============================================================
#  CANNOT BE INSTALLED WITH PIP
#  Set these up manually before running server.py.
# ============================================================
