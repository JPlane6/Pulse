# ============================================================
#  PIP PACKAGES
# ============================================================

requests
# Sends audio data and history JSON structures to your Mac server.

pyserial
# Serial interface link between the Pi and the Arduino Mega motor driver.

RPLCD
# High-level character driver for the 20x4 I2C LCD screen.

smbus2
# Underpinning I2C system communication link for RPLCD layer calls.

sounddevice
# Handles safe 48kHz audio recordings from the USB headset mic.

scipy
# Downsamples audio streams dynamically to 16kHz via signal processing profiles.

numpy
# Array library facilitating multi-channel wave data processing.

opencv-python
# Retained fallback layer for image captures and diagnostic testing.

torch
# Required on the Pi to process local Silero VAD frame computations.

silero-vad
# Dynamic silence threshold iterator to eliminate fixed audio recording caps.

gpiod
# Allows Pi to use GPIO pins to run the status LEDs natively.

#   SUBSYSTEM=="tty", ATTRS{idVendor}=="2341", ATTRS{idProduct}=="0042", SYMLINK+="arduino"
#
# Reload udev system paths:
